cmake_minimum_required(VERSION 3.24)
project(DDSThumbnail VERSION 1.0.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR
   NOT CMAKE_CXX_COMPILER_ARCHITECTURE_ID STREQUAL "x64")
    message(FATAL_ERROR "DDSThumbnail requires Windows and the x64 MSVC toolchain.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

include(FetchContent)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_SAMPLE OFF CACHE BOOL "" FORCE)
set(BUILD_DX11 OFF CACHE BOOL "" FORCE)
set(BUILD_DX12 OFF CACHE BOOL "" FORCE)
set(BC_USE_OPENMP OFF CACHE BOOL "" FORCE)
set(ENABLE_OPENEXR_SUPPORT OFF CACHE BOOL "" FORCE)
set(ENABLE_LIBJPEG_SUPPORT OFF CACHE BOOL "" FORCE)
set(ENABLE_LIBPNG_SUPPORT OFF CACHE BOOL "" FORCE)
FetchContent_Declare(directxtex
    GIT_REPOSITORY https://github.com/microsoft/DirectXTex.git
    GIT_TAG 4feb3e11a020f35b796fc769a74216a555d4f5ef)
FetchContent_MakeAvailable(directxtex)

add_library(dds-render STATIC src/Renderer.cpp src/Stream.cpp)
target_include_directories(dds-render PUBLIC include)
target_link_libraries(dds-render PUBLIC DirectXTex ole32 windowscodecs)
target_compile_definitions(dds-render PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)

add_library(DDSThumbnail SHARED src/Provider.cpp src/Provider.def)
target_link_libraries(DDSThumbnail PRIVATE dds-render gdi32 uuid)
add_executable(dds-preview src/Preview.cpp)
target_link_libraries(dds-preview PRIVATE dds-render shlwapi)

foreach(target dds-render DDSThumbnail dds-preview)
    target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc /utf-8)
endforeach()

include(CTest)
if(BUILD_TESTING)
    add_executable(dds-tests tests/Tests.cpp)
    target_link_libraries(dds-tests PRIVATE dds-render gdi32 shlwapi uuid)
    target_compile_options(dds-tests PRIVATE /W4 /permissive- /EHsc /utf-8)
    add_test(NAME dds-tests COMMAND dds-tests $<TARGET_FILE:DDSThumbnail> ${CMAKE_CURRENT_BINARY_DIR}/fixtures)
    add_executable(dds-shell-test tests/ShellTest.cpp)
    target_link_libraries(dds-shell-test PRIVATE dds-render shell32 shlwapi gdi32 uuid)
    target_compile_options(dds-shell-test PRIVATE /W4 /permissive- /EHsc /utf-8)
endif()

install(TARGETS DDSThumbnail dds-preview RUNTIME DESTINATION . COMPONENT Runtime)
install(FILES scripts/Install.ps1 scripts/Uninstall.ps1 scripts/Common.ps1 README.md VERIFICATION.md DESTINATION . COMPONENT Runtime)
install(FILES ${directxtex_SOURCE_DIR}/LICENSE DESTINATION licenses RENAME DirectXTex-LICENSE.txt COMPONENT Runtime)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "DDSThumbnail-${PROJECT_VERSION}-win-x64")
set(CPACK_ARCHIVE_RUNTIME_FILE_NAME "${CPACK_PACKAGE_FILE_NAME}")
set(CPACK_COMPONENTS_ALL Runtime)
set(CPACK_ARCHIVE_COMPONENT_INSTALL ON)
include(CPack)
